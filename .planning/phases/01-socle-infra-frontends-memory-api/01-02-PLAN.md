class Visibility(str, Enum):
    PRIVATE = "private"; TEAM = "team"; ORG = "org"; PUBLIC = "public"

class TruthLevel(str, Enum):
    EPHEMERAL = "EPHEMERAL"; WORKING = "WORKING"; VALIDATED = "VALIDATED"
    CANONICAL = "CANONICAL"; PUBLIC = "PUBLIC"

class ValidationStatus(str, Enum):
    PENDING = "pending"; VALIDATED = "validated"; REJECTED = "rejected"; NA = "n/a"

class TaggingContract(BaseModel):

    visibility: Visibility
    confidence: float = Field(..., ge=0.0, le=1.0)
    truth_level: TruthLevel = TruthLevel.EPHEMERAL

**`app/deps.py`** — FastAPI dependencies :
```python
from typing import AsyncGenerator
from fastapi import Depends, HTTPException, Header
from sqlalchemy.ext.asyncio import AsyncSession
from app.db.session import async_session_factory
from app.auth import verify_google_id_token, verify_bridge_jwt
from app.config import settings
from app.repos.users import get_or_create_user
from app.repos.teams import get_membership

async def get_session() -> AsyncGenerator[AsyncSession, None]:
    async with async_session_factory() as s:
        yield s

async def get_current_principal(
    authorization: str = Header(...),
    session: AsyncSession = Depends(get_session),
) -> dict:
    if not authorization.startswith("Bearer "):
        raise HTTPException(401, "Missing Bearer token")
    token = authorization.removeprefix("Bearer ")
    # Try Google ID token first, fallback to bridge JWT
    try:
        claims = await verify_google_id_token(token, settings.GOOGLE_CLIENT_ID)
        user = await get_or_create_user(session, source_user_id=claims["sub"], email=claims.get("email", ""), display_name=claims.get("name", ""))
        return {"kind": "user", "user": user, "claims": claims}
    except Exception:
        pass
    try:
        claims = verify_bridge_jwt(token, settings.BRIDGE_SHARED_SECRET)
        return {"kind": "bridge", "claims": claims, "acting_user_sub": claims.get("sub"), "team_scope": claims.get("team_scope")}
    except Exception:
        raise HTTPException(401, "Invalid token")

async def get_team_scope(
    principal: dict = Depends(get_current_principal),
    x_team_scope: str = Header(...),
    session: AsyncSession = Depends(get_session),
) -> str:
    """Vérifie que le principal appartient bien à la team (sauf bridge avec team_scope déjà en claims)."""
    if principal["kind"] == "bridge":
        if principal["team_scope"] != x_team_scope:
            raise HTTPException(403, "Bridge JWT team_scope mismatch")
        return x_team_scope
    user = principal["user"]
    membership = await get_membership(session, user.id, x_team_scope)
    if membership is None:
        raise HTTPException(403, f"Not a member of team {x_team_scope}")
    return x_team_scope
```

```python
from fastapi import APIRouter, Depends, HTTPException, Query
from sqlalchemy.ext.asyncio import AsyncSession
from app.deps import get_session, get_current_principal, get_team_scope
from app.models.tagging import TaggingContract
from app.repos import messages as messages_repo, conversations as conversations_repo
from app.audit import write_audit
from pydantic import BaseModel

router = APIRouter(tags=["messages"])

class MessageCreateBody(BaseModel):
    model_config = {"extra": "forbid"}
    conversation_id: str
    role: str  # validate: "user"|"assistant"|"system"|"tool"
    content: str
    metadata: dict = {}
    # Les 7 champs du contrat de tagging (flat, pas nested — plus simple pour les frontends)
    tagging: TaggingContract  # Pydantic 422 si missing field

class MessageOut(BaseModel):
    id: str
    conversation_id: str
    role: str
    content: str
    tagging: TaggingContract
    created_at: str

@router.post("/messages", status_code=201, response_model=MessageOut)
async def create_message(
    body: MessageCreateBody,
    session: AsyncSession = Depends(get_session),
    principal: dict = Depends(get_current_principal),
    team_scope: str = Depends(get_team_scope),
):
    if body.tagging.team_scope != team_scope:
        raise HTTPException(400, "tagging.team_scope mismatch with X-Team-Scope header")
    conv = await conversations_repo.get_conversation(session, body.conversation_id, team_scope)
    if conv is None:
        raise HTTPException(404, "conversation not found in this team")
    msg = await messages_repo.create_message(session, body.conversation_id, body.tagging, body.role, body.content, body.metadata)
    await session.commit()
    actor = principal.get("user").id if principal["kind"] == "user" else None
    await write_audit(session, actor, team_scope, "messages.create", str(msg.id), {"role": body.role})
    await session.commit()
    return MessageOut(id=str(msg.id), conversation_id=str(msg.conversation_id), role=msg.role, content=msg.content, tagging=body.tagging, created_at=msg.created_at.isoformat())

@router.get("/messages", response_model=list[MessageOut])
async def list_messages(
    session: AsyncSession = Depends(get_session),
    team_scope: str = Depends(get_team_scope),
    conversation_id: str | None = Query(default=None),
    q: str | None = Query(default=None, max_length=200),
    limit: int = Query(default=50, le=200),
):
    rows = await messages_repo.list_messages(session, team_scope=team_scope, conversation_id=conversation_id, q=q, limit=limit)
    return [MessageOut(id=str(m.id), conversation_id=str(m.conversation_id), role=m.role, content=m.content, tagging=TaggingContract(team_scope=m.team_scope, project_scope=m.project_scope, visibility=m.visibility, confidence=m.confidence, truth_level=m.truth_level, source=m.source, validation_status=m.validation_status), created_at=m.created_at.isoformat()) for m in rows]
```

  postgres:
    image: postgres:17
    container_name: xbrain-postgres

  memory-api:

      context: ../apps/memory-api
      dockerfile: Dockerfile
    image: xbrain/memory-api:phase1
    container_name: xbrain-memory-api
    restart: unless-stopped
    environment:
      DATABASE_URL: ${DATABASE_URL}
      QDRANT_URL: ${QDRANT_URL}
