<objective>
Construire `openwebui-pipeline` : un service FastAPI qui présente une API OpenAI-compatible à Open WebUI, proxifie les requêtes vers le vrai LLM (Anthropic/OpenAI), et logge chaque échange (user + assistant) dans memory-api avec le contrat de tagging.

Purpose : Open WebUI n'a pas de webhook outbound natif simple. Le mécanisme officiel est de pointer Open WebUI sur un "OpenAI provider" custom (env `OPENAI_API_BASE_URL`). Notre pipeline est ce provider, transparent pour Open WebUI mais qui logge.

Output : Container Docker, testé, qui complète le pipeline LibreChat→memory-api côté Open WebUI.
</objective>

**`app/memory_api_client.py`** : copier-coller adapté de `apps/librechat-bridge/app/memory_api_client.py`. Différence : `source` arrive dynamiquement (passé par le caller), pas hardcodé. Inclure `make_pipeline_jwt(sub, team_scope, ttl=300)` (équivalent du bridge_token, mais `iss="openwebui-pipeline"`).

```python
import httpx, time
from authlib.jose import jwt
from tenacity import retry, stop_after_attempt, wait_exponential, retry_if_exception_type
from app.config import settings
import structlog

log = structlog.get_logger()

def make_pipeline_jwt(sub: str, team_scope: str, ttl_seconds: int = 300) -> str:
    now = int(time.time())
    return jwt.encode({"alg": settings.JWT_ALGORITHM},
                      {"iss":"openwebui-pipeline","sub":sub,"team_scope":team_scope,
                       "scope":"bridge","iat":now,"exp":now+ttl_seconds},
                      settings.BRIDGE_SHARED_SECRET).decode("ascii")

class MemoryApiClient:
    def __init__(self):
        self.base = settings.MEMORY_API_URL.rstrip("/")
        self.client = httpx.AsyncClient(timeout=10.0)

    @retry(stop=stop_after_attempt(2), wait=wait_exponential(min=0.3, max=2),
           retry=retry_if_exception_type(httpx.HTTPError), reraise=True)
    async def post_message(self, *, sub: str, team_scope: str, conversation_id: str, role: str,
                           content: str, source: str, metadata: dict | None = None) -> dict:
        token = make_pipeline_jwt(sub=sub, team_scope=team_scope)
        body = {"conversation_id": conversation_id, "role": role, "content": content,
                "metadata": metadata or {},
                "tagging": {"team_scope": team_scope, "project_scope": None,
                            "visibility":"team", "confidence":1.0, "truth_level":"EPHEMERAL",
                            "source": source, "validation_status":"pending"}}
        r = await self.client.post(f"{self.base}/v1/messages",
                                   headers={"Authorization": f"Bearer {token}",
                                            "X-Team-Scope": team_scope}, json=body)
        if r.status_code >= 400:
            log.warning("memory_api_post_failed", status=r.status_code, body=r.text[:200])
        r.raise_for_status()
        return r.json()
```

**`app/pipelines/xbrain_logger.py`** :
```python
import asyncio
import structlog
from app.memory_api_client import MemoryApiClient

log = structlog.get_logger()

async def log_exchange(*, mem: MemoryApiClient, sub: str, team_scope: str, conversation_id: str,
                       model: str, user_content: str, assistant_content: str) -> None:
    """Best-effort : on logge mais on n'échoue jamais le call."""
    source = f"openwebui:{model}"
    async def safe_post(role: str, content: str):
        try:
            await mem.post_message(sub=sub, team_scope=team_scope,
                                   conversation_id=conversation_id, role=role,
                                   content=content, source=source)
        except Exception as e:
            log.warning("log_exchange_failed", role=role, err=str(e))
    # Lance les 2 POST en parallèle, ignore les erreurs (best-effort)
    await asyncio.gather(safe_post("user", user_content), safe_post("assistant", assistant_content),
                         return_exceptions=True)
```

**`app/main.py`** : FastAPI exposant 3 endpoints :
- `GET /health` → 200 `{"status":"ok"}`
- `GET /v1/models` → liste de models supportés (Open WebUI les fetch au boot pour son dropdown)
- `POST /v1/chat/completions` → forward vers Anthropic ou OpenAI selon le model, log via xbrain_logger

```python
from fastapi import FastAPI, Header, HTTPException
from fastapi.responses import StreamingResponse
from pydantic import BaseModel
from anthropic import AsyncAnthropic
from openai import AsyncOpenAI
from app.config import settings
from app.memory_api_client import MemoryApiClient
from app.pipelines.xbrain_logger import log_exchange
import asyncio
import structlog

log = structlog.get_logger()
app = FastAPI(title="xbrain openwebui-pipeline", version="0.1.0")
mem = MemoryApiClient()
anthropic_client = AsyncAnthropic(api_key=settings.ANTHROPIC_API_KEY) if settings.ANTHROPIC_API_KEY else None
openai_client = AsyncOpenAI(api_key=settings.OPENAI_API_KEY) if settings.OPENAI_API_KEY else None

# Map "model id" externe → (provider, real_model_name)
MODEL_MAP = {
    "claude-3-5-sonnet": ("anthropic", "claude-3-5-sonnet-latest"),
    "claude-3-5-haiku": ("anthropic", "claude-3-5-haiku-latest"),
    "gpt-4o": ("openai", "gpt-4o"),
    "gpt-4o-mini": ("openai", "gpt-4o-mini"),
}

class ChatMessage(BaseModel):
    role: str
    content: str

class ChatCompletionsBody(BaseModel):
    model: str
    messages: list[ChatMessage]
    stream: bool = False
    user: str | None = None  # Open WebUI passe l'identifiant user ici

def check_auth(authorization: str):
    if not authorization or not authorization.startswith("Bearer "):
        raise HTTPException(401, "Missing api key")
    if authorization.removeprefix("Bearer ") != settings.PIPELINE_API_KEY:
        raise HTTPException(401, "Invalid api key")

@app.get("/health")
async def health(): return {"status": "ok"}

@app.get("/v1/models")
async def list_models(authorization: str = Header(default="")):
    check_auth(authorization)
    return {"object": "list", "data": [{"id": k, "object": "model", "owned_by": "xbrain"} for k in MODEL_MAP]}

@app.post("/v1/chat/completions")
async def chat(body: ChatCompletionsBody,
               authorization: str = Header(default=""),
               x_openwebui_user_id: str | None = Header(default=None),
               x_openwebui_user_email: str | None = Header(default=None),
               x_team_scope: str | None = Header(default=None)):
    check_auth(authorization)
    if body.model not in MODEL_MAP:
        raise HTTPException(400, f"unknown model {body.model}")
    provider, real_model = MODEL_MAP[body.model]
    user_message = next((m.content for m in reversed(body.messages) if m.role == "user"), "")
    sub = x_openwebui_user_id or x_openwebui_user_email or (body.user or "anonymous")
    team_scope = x_team_scope or settings.PIPELINE_DEFAULT_TEAM_SCOPE
    conversation_id = f"openwebui-{sub}-{abs(hash(tuple(m.content for m in body.messages[:1]))) % 10**8}"

    if provider == "anthropic":
        if not anthropic_client: raise HTTPException(500, "Anthropic key not configured")
        if body.stream:
            async def gen():
                full_response = []
                async with anthropic_client.messages.stream(
                    model=real_model, max_tokens=4096,
                    messages=[{"role":m.role,"content":m.content} for m in body.messages if m.role != "system"],
                ) as stream:
                    async for chunk in stream.text_stream:
                        full_response.append(chunk)
                        yield f"data: {{\"choices\":[{{\"delta\":{{\"content\":{chunk!r}}}}}]}}\n\n"
                yield "data: [DONE]\n\n"
                # log après le streaming complet
                asyncio.create_task(log_exchange(mem=mem, sub=sub, team_scope=team_scope,
                                                conversation_id=conversation_id, model=body.model,
                                                user_content=user_message, assistant_content="".join(full_response)))
            return StreamingResponse(gen(), media_type="text/event-stream")
        else:
            r = await anthropic_client.messages.create(model=real_model, max_tokens=4096,
                messages=[{"role":m.role,"content":m.content} for m in body.messages if m.role != "system"])
            assistant_content = "".join([b.text for b in r.content if b.type == "text"])
            asyncio.create_task(log_exchange(mem=mem, sub=sub, team_scope=team_scope,
                                            conversation_id=conversation_id, model=body.model,
                                            user_content=user_message, assistant_content=assistant_content))
            return {"id": f"chatcmpl-{r.id}", "object": "chat.completion",
                    "model": body.model, "choices": [{"index":0,"message":{"role":"assistant","content":assistant_content},"finish_reason":"stop"}]}
    elif provider == "openai":
        if not openai_client: raise HTTPException(500, "OpenAI key not configured")
        # Similar pattern with openai_client.chat.completions.create
        r = await openai_client.chat.completions.create(model=real_model,
            messages=[{"role":m.role,"content":m.content} for m in body.messages])
        assistant_content = r.choices[0].message.content or ""
        asyncio.create_task(log_exchange(mem=mem, sub=sub, team_scope=team_scope,
                                        conversation_id=conversation_id, model=body.model,
                                        user_content=user_message, assistant_content=assistant_content))
        return {"id": r.id, "object": "chat.completion", "model": body.model,
                "choices": [{"index":0,"message":{"role":"assistant","content":assistant_content},"finish_reason":"stop"}]}
```
