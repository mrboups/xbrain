<objective>
Construire le sidecar `librechat-bridge` : un service Python qui watch MongoDB change streams de LibreChat et POST chaque nouveau message sur memory-api avec le contrat de tagging 7 champs.

Purpose : LibreChat n'a pas de webhook outbound. Sans ce bridge, les conversations LibreChat n'atterrissent jamais dans la mémoire commune (success criterion 2 cassé).

Output : Un container Docker autonome qui survit aux restarts, idempotent, et dont les tests verrouillent le mapping payload → contrat.
</objective>

**`app/bridge_token.py`** :
```python
from authlib.jose import jwt
import time
from app.config import settings

def make_bridge_jwt(sub: str, team_scope: str, ttl_seconds: int = 300) -> str:
    """Génère un service JWT court-vif que memory-api va vérifier."""
    now = int(time.time())
    header = {"alg": settings.JWT_ALGORITHM}
    payload = {
        "iss": "librechat-bridge",
        "sub": sub,
        "team_scope": team_scope,
        "scope": "bridge",
        "iat": now,
        "exp": now + ttl_seconds,
    }
    return jwt.encode(header, payload, settings.BRIDGE_SHARED_SECRET).decode("ascii")
```

**`app/memory_api_client.py`** :
```python
import httpx
from tenacity import retry, stop_after_attempt, wait_exponential, retry_if_exception_type
from app.bridge_token import make_bridge_jwt
from app.config import settings
import structlog

log = structlog.get_logger()

class MemoryApiClient:
    def __init__(self):
        self.base = settings.MEMORY_API_URL.rstrip("/")
        self.client = httpx.AsyncClient(timeout=10.0)

    @retry(stop=stop_after_attempt(3), wait=wait_exponential(min=0.5, max=4),
           retry=retry_if_exception_type(httpx.HTTPError), reraise=True)
    async def post_message(self, *, sub: str, team_scope: str, conversation_id: str,
                           role: str, content: str, source: str, metadata: dict | None = None) -> dict:
        token = make_bridge_jwt(sub=sub, team_scope=team_scope)
        body = {
            "conversation_id": conversation_id,
            "role": role,
            "content": content,
            "metadata": metadata or {},
            "tagging": {
                "team_scope": team_scope,
                "project_scope": None,
                "visibility": "team",
                "confidence": 1.0,
                "truth_level": "EPHEMERAL",
                "source": source,
                "validation_status": "pending",
            },
        }
        r = await self.client.post(f"{self.base}/v1/messages",
                                    headers={"Authorization": f"Bearer {token}",
                                             "X-Team-Scope": team_scope},
                                    json=body)
        if r.status_code >= 400:
            log.warning("memory_api_post_failed", status=r.status_code, body=r.text[:200])
        r.raise_for_status()
        return r.json()

    async def post_conversation(self, *, sub: str, team_scope: str, title: str, source: str) -> dict:
        token = make_bridge_jwt(sub=sub, team_scope=team_scope)
        r = await self.client.post(f"{self.base}/v1/conversations",
                                    headers={"Authorization": f"Bearer {token}",
                                             "X-Team-Scope": team_scope},
                                    json={"title": title, "source": source})
        r.raise_for_status()
        return r.json()
```

    try: return json.loads(PATH.read_text())

    PATH.write_text(json.dumps(token, default=str))

**`app/mongo_watcher.py`** : connecte au MongoDB LibreChat via motor, ouvre un change stream sur `messages`, traite chaque insert.

```python
import asyncio
import time
from motor.motor_asyncio import AsyncIOMotorClient
from bson import ObjectId
import structlog
from app.config import settings
from app.memory_api_client import MemoryApiClient
from app.state_store import load_resume_token, save_resume_token

log = structlog.get_logger()

async def resolve_team_scope(client: MemoryApiClient, user_sub: str) -> str:
    """Résout le team_scope d'un user via memory-api /v1/me ou fallback."""
    # Phase 1 simplifié : on appelle pas memory-api pour découvrir le team — on utilise le défaut.
    # En Phase 2 on enrichit avec un cache user→team via /v1/me.
    return settings.BRIDGE_DEFAULT_TEAM_SCOPE

async def map_message(mongo_doc: dict, mongo_db, mem: MemoryApiClient) -> dict | None:
    """Traduit un document Mongo `messages` LibreChat → payload memory-api."""
    conversation_id_lc = mongo_doc.get("conversationId") or mongo_doc.get("parentMessageId")
    if not conversation_id_lc:
        log.info("skip_message_no_convid", _id=str(mongo_doc.get("_id")))
        return None
    convs = mongo_db["conversations"]
    conv = await convs.find_one({"conversationId": conversation_id_lc})
    user_id = mongo_doc.get("user") or (conv.get("user") if conv else None)
    if not user_id:
        log.info("skip_message_no_user", _id=str(mongo_doc.get("_id")))
        return None
    users = mongo_db["users"]
    user = await users.find_one({"_id": ObjectId(user_id)}) if isinstance(user_id, str) else None
    sub = (user or {}).get("googleId") or (user or {}).get("email") or str(user_id)
    model = mongo_doc.get("model") or (conv.get("model") if conv else "unknown")
    source = f"librechat:{model}"
    role = "user" if mongo_doc.get("isCreatedByUser") else "assistant"
    content = mongo_doc.get("text") or ""
    return {"sub": sub, "conversation_id": conversation_id_lc, "role": role, "content": content,
            "source": source, "metadata": {"librechat_id": str(mongo_doc.get("_id"))}}

async def watch_loop():
    mongo = AsyncIOMotorClient(settings.LIBRECHAT_MONGO_URI)
    db = mongo.get_default_database()
    mem = MemoryApiClient()
    resume_token = load_resume_token()

    pipeline = [{"$match": {"operationType": "insert", "ns.coll": "messages"}}]
    log.info("starting_change_stream", resume=bool(resume_token))
    async with db.watch(pipeline=pipeline, resume_after=resume_token) as stream:
        async for change in stream:
            try:
                doc = change["fullDocument"]
                payload = await map_message(doc, db, mem)
                if payload:
                    team_scope = await resolve_team_scope(mem, payload["sub"])
                    # Garantir qu'une conv existe côté memory-api (idempotent côté serveur — Phase 2 fera UPSERT)
                    await mem.post_message(team_scope=team_scope, **payload)
                    log.info("forwarded_message", lc_id=payload["metadata"]["librechat_id"], team=team_scope)
                save_resume_token(change.get("_id"))
                # heartbeat for healthcheck
                with open(settings.BRIDGE_HEARTBEAT_PATH, "w") as f: f.write(str(int(time.time())))
            except Exception as e:
                log.error("watch_event_failed", err=str(e))
                # Don't crash the whole watcher on a single bad event
                continue

async def heartbeat_loop():
    """Touche le heartbeat même si Mongo est silencieux, pour que docker healthcheck passe."""
    while True:
        with open(settings.BRIDGE_HEARTBEAT_PATH, "w") as f: f.write(str(int(time.time())))
        await asyncio.sleep(20)
```

import asyncio

import logging
