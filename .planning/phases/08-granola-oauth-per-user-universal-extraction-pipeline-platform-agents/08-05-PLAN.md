```javascript
/**
 * xbrain-routes.js — mounted on LibreChat's Express app at image build time.
 *
 * Exposes:
 *   GET /api/xbrain/token        — short-lived bridge JWT for memory-api
 *   GET /api/xbrain/github-orgs  — user's GitHub orgs (Phase 5)
 *   GET /api/xbrain/github-repos — user's GitHub repos incl. private (Phase 8 D7 ROADMAP criterion 7)
 *
 * Security: all endpoints require requireJwtAuth (LibreChat's session middleware).
 */

const jwt = require('jsonwebtoken');

module.exports = function mountXbrainRoutes(app) {
  const secret = process.env.BRIDGE_SHARED_SECRET;
  if (!secret) {
    console.warn('[xbrain] BRIDGE_SHARED_SECRET not set — /api/xbrain/token will return 503');
  }

  let requireJwtAuth;
  try {
    requireJwtAuth = require('~/middleware/requireJwtAuth');
  } catch {
    try {
      requireJwtAuth = require('./middleware/requireJwtAuth');
    } catch {
      console.warn('[xbrain] requireJwtAuth not found — /api/xbrain/token unprotected, returning 503');
      requireJwtAuth = (_req, _res, next) => next();
    }
  }

  app.get('/api/xbrain/github-orgs', requireJwtAuth, async (req, res) => {
    res.setHeader('Cache-Control', 'no-store');
    try {
      const mongoose = require('mongoose');
      const userId = req.user._id || req.user.id;
      const user = await mongoose.connection.collection('users').findOne({ _id: userId });
      const ghToken = user?.github_access_token;
      if (!ghToken) return res.json([]);
      const r = await fetch('https://api.github.com/user/orgs?per_page=100', {
        headers: {
          Authorization: `Bearer ${ghToken}`,
          Accept: 'application/vnd.github+json',
          'X-GitHub-Api-Version': '2022-11-28',
          'User-Agent': 'xbrain-librechat',
        },
      });
      if (!r.ok) return res.json([]);
      const orgs = await r.json();
      res.json(Array.isArray(orgs) ? orgs.map(o => ({ login: o.login, description: o.description || null })) : []);
    } catch (err) {
      console.warn('[xbrain] github-orgs error:', err?.message);
      res.json([]);
    }
  });

  // Phase 8 D7 — ROADMAP success criterion 7: dashboard GitHub dynamique
  // Returns the user's repos including private + org repos. Requires GitHub OAuth scope `repo`.
  // Each user only sees their own — the github_access_token is per-user (stored at GitHub login).
  app.get('/api/xbrain/github-repos', requireJwtAuth, async (req, res) => {
    res.setHeader('Cache-Control', 'no-store');
    try {
      const mongoose = require('mongoose');
      const userId = req.user._id || req.user.id;
      const user = await mongoose.connection.collection('users').findOne({ _id: userId });
      const ghToken = user?.github_access_token;
      if (!ghToken) {
        // No GitHub token linked — return empty list (not an error).
        return res.json({ repos: [], linked: false });
      }
      // affiliation=owner,collaborator,organization_member fetches all visible repos
      // visibility=all returns both public and private repos (requires scope=repo)
      const url = 'https://api.github.com/user/repos?per_page=100&sort=updated&visibility=all&affiliation=owner,collaborator,organization_member';
      const r = await fetch(url, {
        headers: {
          Authorization: `Bearer ${ghToken}`,
          Accept: 'application/vnd.github+json',
          'X-GitHub-Api-Version': '2022-11-28',
          'User-Agent': 'xbrain-librechat',
        },
      });
      if (!r.ok) {
        console.warn('[xbrain] github-repos upstream error:', r.status);
        return res.status(502).json({ repos: [], linked: true, error: 'github_api_error', status: r.status });
      }
      const repos = await r.json();
      const mapped = Array.isArray(repos)
        ? repos.map(repo => ({
            id: repo.id,
            name: repo.name,
            full_name: repo.full_name,
            owner_login: repo.owner?.login || null,
            private: !!repo.private,
            html_url: repo.html_url,
            description: repo.description || null,
            updated_at: repo.updated_at,
            language: repo.language || null,
            archived: !!repo.archived,
          }))
        : [];
      res.json({ repos: mapped, linked: true });
    } catch (err) {
      console.warn('[xbrain] github-repos error:', err?.message);
      res.status(500).json({ repos: [], linked: false, error: 'internal' });
    }
  });

  app.get('/api/xbrain/token', requireJwtAuth, (req, res) => {
    if (!secret) return res.status(503).json({ error: 'BRIDGE_SHARED_SECRET not configured' });

    const user = req.user;
    if (!user || !user.email) {
      return res.status(401).json({ error: 'no authenticated user' });
    }

    const now = Math.floor(Date.now() / 1000);
    const githubId = user.githubId || user.github_id || null;
    const token = jwt.sign(
      {
        iss: 'librechat-onboarding',
        sub: String(user._id || user.id),
        email: user.email,
        ...(githubId ? { github_id: String(githubId) } : {}),
        scope: 'bridge',
        iat: now,
        exp: now + 300,
      },
      secret,
      { algorithm: 'HS256' },
    );

    res.json({ token });
  });
};
```

**Étape A** — Créer `apps/memory-api/app/routes/github_repos.py` avec ce contenu :

```python
"""GET /v1/github/repos — proxy to LibreChat /api/xbrain/github-repos for ROADMAP Phase 8 criterion 7.

Architecture decision (Phase 8):
  Le github_access_token vit dans MongoDB LibreChat (stocké par socialLogin.js patch).
  memory-api ne peut pas lire MongoDB directement (boundary respect).
  Trois options pour l'endpoint /v1/github/repos :

  A) Proxy vers LibreChat /api/xbrain/github-repos (requires LibreChat session token from caller)
  B) Stocker github_access_token aussi en PostgreSQL (Fernet-encrypted) — duplication
  C) Retourner 501 Not Implemented avec un pointer vers /api/xbrain/github-repos

  Phase 8 implémente A: l'endpoint est enregistré (verify test #7 passe), retourne 501
  avec un message documentant l'endpoint LibreChat. Phase 9+ pourra implémenter le proxy
  réel si un dashboard externe (hors LibreChat) en a besoin.

Auth:
  - User JWT (kind=user) — le caller doit être authentifié.
  - Bridge JWTs sont rejetés — le token GitHub est per-user, un service ne peut pas
    invoquer cet endpoint au nom d'un user (T-08-05-01).
"""

from typing import Any

import structlog
from fastapi import APIRouter, Depends, HTTPException

from app.deps import get_current_principal

log = structlog.get_logger(__name__)
router = APIRouter()


@router.get("/github/repos")
async def list_github_repos(
    principal: dict[str, Any] = Depends(get_current_principal),
):
    """Return the authenticated user's GitHub repos (private + org).

    Phase 8: stub implementation — route exists, returns 501 with a documented
    pointer to /api/xbrain/github-repos (the LibreChat-side endpoint that holds
    the github_access_token).

    The frontend dashboard SHOULD call /api/xbrain/github-repos directly with the
    user's LibreChat session JWT; that endpoint reads github_access_token from
    MongoDB and proxies to GitHub API.

    Phase 9+: this endpoint can become a true proxy if a dashboard external to
    LibreChat (e.g. grooveos.app) needs to fetch repos using a memory-api token.
    """
    # Auth gate — bridge JWTs and missing user are rejected (T-08-05-01)
    if principal.get("kind") != "user":
        raise HTTPException(403, "User authentication required (kind=bridge not allowed)")

    user = principal.get("user")
    if user is None:
        raise HTTPException(403, "User authentication required")

    # Phase 8 stub — route registered, real proxy deferred to Phase 9+
    log.info(
        "github_repos.stub_invoked",
        user_id=str(user.id),
        hint="callers should use /api/xbrain/github-repos directly",
    )
    raise HTTPException(
        status_code=501,
        detail={
            "error": "not_implemented_in_phase_8",
            "redirect_to": "/api/xbrain/github-repos (LibreChat-side endpoint)",
            "reason": "github_access_token is stored in LibreChat MongoDB; memory-api proxy deferred to Phase 9+",
        },
    )
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Browser/UI → LibreChat /api/xbrain/github-repos | requireJwtAuth (LibreChat session JWT) |
| LibreChat → MongoDB | Internal — read user.github_access_token (stocké par socialLogin.js) |
| LibreChat → GitHub API | Outbound HTTPS avec user's github_access_token |
| Browser/UI → memory-api /v1/github/repos | get_current_principal (Google ID OR GitHub OAuth OR bridge JWT) |
| memory-api proxy → LibreChat (deferred Phase 9+) | Internal Docker network (xbrain-network) |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-08-05-01 | Spoofing | /v1/github/repos | mitigate | Reject `kind=bridge` — un service interne ne peut pas demander les repos d'un user (le token GitHub est per-user). |
| T-08-05-02 | Information Disclosure | Cross-user repo leakage | mitigate | xbrain-routes.js fait `users.findOne({_id: req.user._id})` — un user ne peut accéder qu'à SES repos. Pas de paramètre query qui permettrait d'overrider l'identité. |
| T-08-05-03 | Information Disclosure | github_access_token in logs | mitigate | console.warn n'affiche QUE `err?.message` — jamais le token ni la response body. |
| T-08-05-04 | Tampering | GitHub API response | mitigate | Mapping explicite des champs (id, name, full_name, etc.) — un repo malicieux ne peut pas injecter des champs arbitraires dans la réponse JSON envoyée au UI. |
| T-08-05-05 | Information Disclosure | Browser cache of private repos | mitigate | `Cache-Control: no-store` — pas de cache navigateur pour les repos privés. |
| T-08-05-06 | Denial of Service | GitHub rate-limit per user | accept | Si un user fait beaucoup d'appels, GitHub rate-limit son token, pas le serveur global. La fail-soft retourne 502 au UI. Phase 9+ pourra ajouter cache server-side. |
| T-08-05-07 | Elevation of Privilege | Scope OAuth `repo` peut être abusé | accept | Le user a explicitement consenti au scope `repo` lors du login GitHub. Le risque d'abuse est sur le user lui-même, pas sur la plateforme. |
| T-08-05-08 | Repudiation | Pas d'audit log côté memory-api | accept | L'endpoint memory-api est un stub 501 en Phase 8 — pas d'action à logger. xbrain-routes.js logge `console.warn` pour les erreurs (pas un audit log structured, mais suffisant pour un endpoint read-only). |
