  function renderGranolaConnected(statusData) {
    const body = getBody();
    const stepLabel = el('p', { class: 'xb-step-indicator' }, 'GRANOLA CONNECTION');
    const title = el('h2', {}, '✓ Granola connecté');

    const teamScope = (statusData && statusData.team_scope) || (selectedTeam?.slug) || '';
    const desc = el('p', { class: 'xb-desc' },
      `Your Granola key is active${teamScope ? ' for team "' + teamScope + '"' : ''}. ` +
      'New meetings will be polled and added to your team\'s memory automatically. ' +
      'You can disconnect at any time — your existing meetings will remain in memory.');

    const errorDiv = el('p', { class: 'xb-error', style: 'display:none' });
    const successDiv = el('p', { class: 'xb-desc', style: 'color:#7fc08f;display:none' });

    const closeBtn = el('button', { class: 'xb-btn xb-btn-primary' }, 'Close');
    closeBtn.onclick = dismiss;

    const disconnectBtn = el('button', { class: 'xb-btn xb-btn-secondary' }, 'Déconnecter');
    disconnectBtn.onclick = async () => {
      errorDiv.style.display = 'none';
      // Confirm action — pas un confirm() bloquant, juste désactiver pour éviter double-click
      disconnectBtn.disabled = true;
      closeBtn.disabled = true;
      // Pitfall 7 — fresh token juste avant l'appel
      const freshToken = await getToken();
      if (!freshToken) {
        errorDiv.textContent = 'Authentication expired. Please refresh and try again.';
        errorDiv.style.display = 'block';
        disconnectBtn.disabled = false;
        closeBtn.disabled = false;
        return;
      }
      try {
        // DELETE /v1/me/granola-key — révoque la clé (criterion 1 ROADMAP).
        const r = await apiCall('DELETE', '/v1/me/granola-key', null, freshToken);
        if (!r.ok && r.status !== 204) {
          const errBody = await r.json().catch(() => ({}));
          throw new Error(errBody.detail || `HTTP ${r.status}`);
        }
        // Success — affiche feedback + bouton "Reconnect" (revient au mode connect)
        successDiv.textContent = 'Granola déconnecté.';
        successDiv.style.display = 'block';
        title.textContent = 'Granola déconnecté';
        desc.textContent = 'Your Granola key has been removed. You can reconnect at any time.';
        disconnectBtn.style.display = 'none';

        body.appendChild(reconnectBtn);
        closeBtn.disabled = false;
      } catch (e) {
        errorDiv.textContent = `Could not disconnect: ${e.message || 'unknown error'}`;
        errorDiv.style.display = 'block';
        disconnectBtn.disabled = false;
        closeBtn.disabled = false;
      }
    };

    setHtml(body, stepLabel, title, desc, errorDiv, successDiv, closeBtn, disconnectBtn);
  }

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Browser → /api/xbrain/token | LibreChat session JWT (httpOnly cookie ou XHR Authorization) |
| Browser → memory-api /v1/me/granola-key | Bridge JWT (iss=librechat-onboarding, exp=5min) signed by BRIDGE_SHARED_SECRET |
| Browser DOM → memory in clear | api_key visible during typing (input type=password masque visuellement) |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-08-07-01 | Information Disclosure | api_key in browser DOM | mitigate | type=password masque la value à l'écran. autocomplete=off — pas de cache navigateur. |
| T-08-07-02 | Tampering | Replay LibreChat session token | mitigate | Pitfall 7 — getToken() refresh JUST avant chaque submit (POST + DELETE). Le bridge JWT issued a exp=5min — fenêtre courte. |
| T-08-07-03 | Spoofing | Cross-user submission/disconnection | mitigate | Le bridge JWT iss=librechat-onboarding contient le sub réel du LibreChat user. Le DELETE supprime la row WHERE user_id = sub — impossible de cibler un autre user. |
| T-08-07-04 | Information Disclosure | api_key in browser console.log | accept | Pas de console.log de la clé dans le code. Si l'utilisateur ouvre devtools et logge manuellement, c'est son own-goal. |
| T-08-07-05 | Repudiation | Connexion/déconnexion silencieuse | mitigate | POST/DELETE /v1/me/granola-key écrivent audit_log (`me.granola_key.upserted` / `me.granola_key.revoked`) côté memory-api (plan 08-03). |
| T-08-07-06 | Denial of Service | XSS via api_key field | mitigate | Le contenu de api_key est envoyé en JSON body — pas inséré dans le DOM. errorDiv.textContent (pas innerHTML) — pas d'injection HTML possible via error message. |
| T-08-07-07 | Tampering | Bridge JWT secret leak via JS | accept | Le BRIDGE_SHARED_SECRET n'est JAMAIS dans le JS. Le JS ne fait que demander un bridge JWT à `/api/xbrain/token` — c'est le serveur LibreChat qui signe. |
| T-08-07-08 | Repudiation | Disconnect accidentel | accept | Pas de confirm() bloquant — le user peut Reconnect immédiatement. Risque accepté pour simplicité UX. |
</threat_model>
