**apps/memory-api/app/routes/admin_drive.py** :
```python
"""/v1/admin/drive-mapping — admin endpoints for Drive folder → team mapping.

Flow:
  1. Admin calls POST /v1/admin/drive-mapping with {team_scope, folder_id}
  2. Endpoint creates/updates row in team_drive_mappings and returns {authorization_url}
  3. Admin (or user on behalf of team) visits authorization_url in browser
  4. Google redirects to /v1/admin/drive-mapping/oauth-callback?code=...&state=team_scope
  5. Callback exchanges code for tokens, encrypts with Fernet, stores in team_drive_mappings
"""
from __future__ import annotations

import json
from typing import Any

import httpx
import structlog
from fastapi import APIRouter, Depends, HTTPException
from pydantic import BaseModel

from app.config import settings
from app.deps import get_current_principal, get_session

log = structlog.get_logger(__name__)
router = APIRouter()


def _require_fernet():
    if not settings.OAUTH_CREDENTIALS_ENCRYPTION_KEY:
        raise HTTPException(500, "OAUTH_CREDENTIALS_ENCRYPTION_KEY not configured — cannot store OAuth credentials")
    from cryptography.fernet import Fernet
    return Fernet(settings.OAUTH_CREDENTIALS_ENCRYPTION_KEY.encode())


def _is_admin(principal: dict[str, Any]) -> bool:
    """Allow bridge service JWTs (scope=bridge) or listed admin subs."""
    if principal.get("kind") == "service":
        return True
    sub = principal.get("sub", "")
    admin_subs = [s.strip() for s in (settings.ADMIN_USER_SUBS or "").split(",") if s.strip()]
    return sub in admin_subs


class DriveMappingBody(BaseModel):
    team_scope: str
    folder_id: str


@router.post("/admin/drive-mapping", status_code=201)
async def create_drive_mapping(
    body: DriveMappingBody,
    session=Depends(get_session),
    principal: dict[str, Any] = Depends(get_current_principal),
):
    """Create or update a Drive folder → team mapping.
    
    Returns an authorization_url the user must visit to grant drive.readonly OAuth consent.
    """
    if not _is_admin(principal):
        raise HTTPException(403, "Admin access required")
    if not settings.GOOGLE_CLIENT_ID:
        raise HTTPException(500, "GOOGLE_CLIENT_ID not configured")

    # Upsert into team_drive_mappings
    await session.execute(
        """INSERT INTO team_drive_mappings(team_scope, folder_id)
           VALUES(:team_scope, :folder_id)
           ON CONFLICT(team_scope) DO UPDATE
           SET folder_id=EXCLUDED.folder_id, updated_at=now()""",
        {"team_scope": body.team_scope, "folder_id": body.folder_id},
    )
    await session.commit()

    redirect_uri = f"{settings.MEMORY_API_EXTERNAL_URL}/v1/admin/drive-mapping/oauth-callback"
    authorization_url = (
        "https://accounts.google.com/o/oauth2/v2/auth"
        f"?client_id={settings.GOOGLE_CLIENT_ID}"
        f"&redirect_uri={redirect_uri}"
        "&scope=https%3A%2F%2Fwww.googleapis.com%2Fauth%2Fdrive.readonly"
        "&include_granted_scopes=true"
        "&access_type=offline"
        "&response_type=code"
        f"&state={body.team_scope}"
        "&prompt=consent"
    )
    log.info("drive_mapping.created", team=body.team_scope, folder=body.folder_id)
    return {
        "team_scope": body.team_scope,
        "folder_id": body.folder_id,
        "authorization_url": authorization_url,
        "next_step": "Visit authorization_url in browser to grant Drive read access",
    }


@router.get("/admin/drive-mapping/{team_scope}")
async def get_drive_mapping(
    team_scope: str,
    session=Depends(get_session),
    principal: dict[str, Any] = Depends(get_current_principal),
):
    """Get the Drive mapping for a team (omits encrypted credentials)."""
    if not _is_admin(principal):
        raise HTTPException(403, "Admin access required")
    row = await session.execute(
        "SELECT team_scope, folder_id, change_token, updated_at FROM team_drive_mappings WHERE team_scope=:ts",
        {"ts": team_scope},
    )
    result = row.fetchone()
    if result is None:
        raise HTTPException(404, f"No Drive mapping for team '{team_scope}'")
    return {
        "team_scope": result.team_scope,
        "folder_id": result.folder_id,
        "change_token": result.change_token,
        "updated_at": str(result.updated_at),
        "oauth_configured": True,  # We don't expose the encrypted creds
    }


@router.get("/admin/drive-mapping/oauth-callback")
async def drive_oauth_callback(
    code: str,
    state: str,
    session=Depends(get_session),
):
    """Google OAuth callback — exchange code for tokens and store encrypted.
    
    state = team_scope (set in the authorization URL).
    This endpoint is called by Google after the user grants consent.
    It is intentionally unauthenticated (Google redirects here with the code).
    """
    redirect_uri = f"{settings.MEMORY_API_EXTERNAL_URL}/v1/admin/drive-mapping/oauth-callback"
    fernet = _require_fernet()

    # Exchange code for tokens
    async with httpx.AsyncClient(timeout=15.0) as client:
        resp = await client.post(
            "https://oauth2.googleapis.com/token",
            data={
                "code": code,
                "client_id": settings.GOOGLE_CLIENT_ID,
                "client_secret": settings.GOOGLE_CLIENT_SECRET,
                "redirect_uri": redirect_uri,
                "grant_type": "authorization_code",
            },
        )
        resp.raise_for_status()
        tokens = resp.json()

    # Encrypt and store
    encrypted = fernet.encrypt(json.dumps(tokens).encode()).decode()
    team_scope = state
    await session.execute(
        "UPDATE team_drive_mappings SET oauth_credentials_enc=:enc, updated_at=now() WHERE team_scope=:ts",
        {"enc": encrypted, "ts": team_scope},
    )
    await session.commit()
    log.info("drive_oauth.callback_success", team=team_scope)
    return {"status": "OAuth credentials stored successfully", "team_scope": team_scope}
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Admin → POST /admin/drive-mapping | Requiert bridge JWT ou sub listé dans ADMIN_USER_SUBS |
| Google → /admin/drive-mapping/oauth-callback | Callback unauthentifié par design (Google n'envoie pas de JWT) — code valide une fois |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-10-01 | Elevation of Privilege | oauth-callback unauthentifié | accept | Le `code` OAuth Google est single-use et expire en 10min. L'`state` param contient team_scope mais n'est pas secret — juste un identifiant. Attaque CSRF nécessiterait d'intercepter le code Google. |
| T-03-10-02 | Information Disclosure | oauth_credentials_enc en DB | mitigate | Chiffrement Fernet — la clé est dans OAUTH_CREDENTIALS_ENCRYPTION_KEY (env var, pas en DB). GET /admin/drive-mapping omet les credentials. |
| T-03-10-03 | Tampering | folder_id arbitraire | accept | Admin-only endpoint. L'admin choisit le folder — c'est son privilège. |
</threat_model>
