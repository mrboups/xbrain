```bash
#!/usr/bin/env bash
# register-mcp-tools.sh — Register the 3 Phase 3 MCP sidecars in mcp-gateway.
#
# Usage: bash infrastructure/scripts/register-mcp-tools.sh
#
# Requires: jq, curl, BRIDGE_SHARED_SECRET env var set.
# The script generates a bridge JWT internally via Python (available in mcp-gateway container).
# Run this script AFTER all 5 MCP containers are healthy.
#
# Idempotent: POST /admin/register uses ON CONFLICT DO UPDATE — safe to re-run.

set -euo pipefail

MCP_GATEWAY_URL="${MCP_GATEWAY_URL:-http://localhost:8080}"
BRIDGE_SHARED_SECRET="${BRIDGE_SHARED_SECRET:?BRIDGE_SHARED_SECRET env var required}"

echo "=== xbrain MCP Tool Registration ==="
echo "Gateway: $MCP_GATEWAY_URL"

# Generate a bridge JWT via Python
BRIDGE_JWT=$(python3 -c "
import time, sys
try:
    from authlib.jose import jwt as jose_jwt
except ImportError:
    print('ERROR: authlib not installed. Run: pip install authlib', file=sys.stderr)
    sys.exit(1)
secret = '$BRIDGE_SHARED_SECRET'
payload = {'sub': 'admin-script', 'scope': 'bridge', 'iat': int(time.time()), 'exp': int(time.time()) + 3600}
token = jose_jwt.encode({'alg': 'HS256'}, payload, secret).decode()
print(token)
")

echo "Bridge JWT generated (expires in 1h)"

# Helper function
register_tool() {
    local name="$1"
    local url="$2"
    local desc="$3"

    echo ""
    echo "Registering: $name → $url"
    response=$(curl -sf -X POST "$MCP_GATEWAY_URL/admin/register" \
        -H "Content-Type: application/json" \
        -H "Authorization: Bearer $BRIDGE_JWT" \
        -d "{\"tool_name\": \"$name\", \"sidecar_url\": \"$url\", \"description\": \"$desc\"}" \
        2>&1) || { echo "  FAILED: $response"; return 1; }
    echo "  OK: $response" | head -c 200
}

# Register the 3 Phase 3 tools
register_tool "scraper" "http://mcp-scraper:8100" "Fetch URL content as text (max 50KB) — MCP-05"
register_tool "drive-read" "http://mcp-drive-read:8101" "Live read/write a Google Drive file by ID — MCP-05, INT-04"
register_tool "calendar" "http://mcp-calendar:8102" "List Google Calendar events for a date range — MCP-06"

echo ""
echo "=== Verification ==="
echo "Registered tools:"
curl -sf "$MCP_GATEWAY_URL/tools" \
    -H "Authorization: Bearer $BRIDGE_JWT" | python3 -m json.tool 2>/dev/null || echo "(jq/json not available)"

echo ""
echo "=== Test scraper tool ==="
echo "Testing: POST /tools/scraper/call"
curl -sf -X POST "$MCP_GATEWAY_URL/tools/scraper/call" \
    -H "Content-Type: application/json" \
    -H "Authorization: Bearer $BRIDGE_JWT" \
    -H "X-Team-Scope: default" \
    -d '{"jsonrpc": "2.0", "id": 1, "method": "tools/call", "params": {"name": "scraper", "arguments": {"url": "https://example.com"}}}' \
    | python3 -c "import sys,json; d=json.load(sys.stdin); print('SCRAPER OK — chars:', len(str(d)))" \
    || echo "Scraper test failed (expected if mcp-scraper not healthy yet)"

echo ""
echo "=== Done ==="
echo "To test from agent-runtime (inside the Docker network), use http://mcp-gateway:8080"
echo "To test from host (port forwarding required), use http://localhost:8080"
echo ""
echo "Manual test commands:"
echo "  # List tools:"
echo "  curl -H 'Authorization: Bearer \$JWT' http://mcp-gateway:8080/tools"
echo ""
echo "  # Call scraper:"
echo "  curl -X POST http://mcp-gateway:8080/tools/scraper/call \\"
echo "    -H 'Authorization: Bearer \$JWT' -H 'X-Team-Scope: default' \\"
echo "    -d '{\"jsonrpc\":\"2.0\",\"id\":1,\"method\":\"tools/call\",\"params\":{\"name\":\"scraper\",\"arguments\":{\"url\":\"https://example.com\"}}}'"
echo ""
echo "  # Call calendar (requires OAuth setup — see docs/google-oauth-scope-upgrade.md):"
echo "  curl -X POST http://mcp-gateway:8080/tools/calendar/call \\"
echo "    -H 'Authorization: Bearer \$JWT' -H 'X-Team-Scope: default' \\"
echo "    -d '{\"jsonrpc\":\"2.0\",\"id\":1,\"method\":\"tools/call\",\"params\":{\"name\":\"calendar\",\"arguments\":{\"date_range\":\"today+7days\"}}}'"
```
