```python
"""Truth-level promotion state machine.

Rules:
- EPHEMERAL → WORKING : auto, no approval (triggered by N>=3 messages in conv)
- WORKING → VALIDATED : 1 team admin approves
- VALIDATED → CANONICAL : 2 distinct team admins approve (4-eyes)
- CANONICAL → PUBLIC : 1 org admin approves
- {any} → REJECTED : back to WORKING with validation_status='rejected'
"""
from enum import Enum
from typing import Any
from uuid import UUID
from sqlalchemy.ext.asyncio import AsyncSession
from xbrain_memory import MemoryProvider, TruthLevel
from app.audit import write_audit
from app.repos import promotions as promo_repo


ALLOWED_TRANSITIONS = {
    (TruthLevel.EPHEMERAL, TruthLevel.WORKING),
    (TruthLevel.WORKING,   TruthLevel.VALIDATED),
    (TruthLevel.VALIDATED, TruthLevel.CANONICAL),
    (TruthLevel.CANONICAL, TruthLevel.PUBLIC),
    # rejections — return to WORKING from any level (enforce status='rejected')
}

# Number of distinct admin approvals required per transition
APPROVAL_REQUIREMENTS = {
    (TruthLevel.EPHEMERAL, TruthLevel.WORKING): 0,    # auto
    (TruthLevel.WORKING,   TruthLevel.VALIDATED): 1,
    (TruthLevel.VALIDATED, TruthLevel.CANONICAL): 2,  # 4-eyes
    (TruthLevel.CANONICAL, TruthLevel.PUBLIC): 1,     # org admin
}


def validate_transition(from_level: TruthLevel, to_level: TruthLevel) -> None:
    if (from_level, to_level) not in ALLOWED_TRANSITIONS:
        raise ValueError(f"Illegal transition {from_level} → {to_level}")


def required_approvers(from_level: TruthLevel, to_level: TruthLevel) -> int:
    return APPROVAL_REQUIREMENTS.get((from_level, to_level), 1)


async def propose_promotion(
    session: AsyncSession,
    *,
    memory_item_id: UUID,
    team_scope: str,
    from_level: TruthLevel,
    to_level: TruthLevel,
    proposed_by: UUID,
    rationale: str | None,
) -> UUID:
    validate_transition(from_level, to_level)
    promo_id = await promo_repo.create_promotion(
        session, memory_item_id=memory_item_id, team_scope=team_scope,
        from_level=from_level.value, to_level=to_level.value,
        proposed_by=proposed_by, rationale=rationale,
    )
    await write_audit(session, actor_user_id=proposed_by, team_scope=team_scope,
                      action="promotions.propose", target_id=str(promo_id),
                      payload={"from": from_level.value, "to": to_level.value})
    return promo_id


async def approve_promotion(
    session: AsyncSession, provider: MemoryProvider,
    *,
    promotion_id: UUID,
    approver: UUID,
    is_admin: bool,
) -> dict[str, Any]:
    if not is_admin:
        raise PermissionError("admin only")
    promo = await promo_repo.get_promotion(session, promotion_id)
    if promo is None or promo.status != "pending":
        raise ValueError("promotion not pending")
    if promo.proposed_by == approver:
        raise PermissionError("proposer can't be approver (separation of duty)")
    needed = required_approvers(TruthLevel(promo.from_level), TruthLevel(promo.to_level))

    # Record approval
    if not promo.approved_by_1:
        promo.approved_by_1 = approver
    elif not promo.approved_by_2 and promo.approved_by_1 != approver:
        promo.approved_by_2 = approver
    else:
        raise ValueError("approver already recorded OR no slot left")

    approvals_so_far = sum(1 for x in [promo.approved_by_1, promo.approved_by_2] if x)
    if approvals_so_far >= needed:
        # Apply transition: update memory_items.truth_level via provider
        item = await provider.get(str(promo.memory_item_id), team_scope=promo.team_scope)
        if item is None:
            raise ValueError("memory item missing")
        await provider.update(str(promo.memory_item_id), team_scope=promo.team_scope,
                              patch={"truth_level": TruthLevel(promo.to_level)})
        promo.status = "approved"
        promo.resolved_at = datetime.now(timezone.utc)
        await write_audit(session, actor_user_id=approver, team_scope=promo.team_scope,
                          action="promotions.approved",
                          target_id=str(promo.id),
                          payload={"to_level": promo.to_level, "approvals": approvals_so_far})
    await session.commit()
    return {"promotion_id": str(promo.id), "status": promo.status,
            "approvals": approvals_so_far, "needed": needed}
```

```python
"""OWUI Pipeline 'promotion-manager' — admin commands inside the chat."""
import re
from typing import Any
import structlog

from app.memory_api_client import MemoryApiClient

log = structlog.get_logger()

# Commands matched in chat input
PENDING_RE = re.compile(r"^/promotions-pending\s*$")
APPROVE_RE = re.compile(r"^/approve\s+([a-f0-9-]{36})\s*$")
REJECT_RE = re.compile(r"^/reject\s+([a-f0-9-]{36})\s+(.+)$")


async def handle_command(text: str, *, mem: MemoryApiClient, sub: str, team_scope: str) -> str | None:
    """Returns Markdown reply if the text is a promotion command, else None."""
    if PENDING_RE.match(text):
        promos = await mem.list_pending_promotions(sub=sub, team_scope=team_scope)
        if not promos:
            return "*No pending promotions*"
        rows = [f"| ID | from→to | Rationale |\n|---|---|---|"]
        rows += [f"| `{p['id']}` | {p['from_level']} → {p['to_level']} | {p.get('rationale','')[:80]} |" for p in promos]
        return "\n".join(rows)
    if m := APPROVE_RE.match(text):
        promo_id = m.group(1)
        result = await mem.approve_promotion(sub=sub, team_scope=team_scope, promotion_id=promo_id)
        return f"✅ Approved `{promo_id}` — status: {result['status']} ({result['approvals']}/{result['needed']} approvals)"
    if m := REJECT_RE.match(text):
        promo_id, reason = m.group(1), m.group(2)
        await mem.reject_promotion(sub=sub, team_scope=team_scope, promotion_id=promo_id, reason=reason)
        return f"❌ Rejected `{promo_id}` — reason: {reason}"
    return None
```
