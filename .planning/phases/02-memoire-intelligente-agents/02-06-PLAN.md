"""LangGraph PostgresSaver factory (singleton)."""
from langgraph.checkpoint.postgres.aio import AsyncPostgresSaver
from psycopg_pool import AsyncConnectionPool
from app.config import settings

_pool: AsyncConnectionPool | None = None
_saver: AsyncPostgresSaver | None = None


async def get_checkpointer() -> AsyncPostgresSaver:
    global _pool, _saver
    if _saver is not None:
        return _saver
    # Convert SQLAlchemy URL to plain Postgres
    dsn = settings.DATABASE_URL.replace("postgresql+asyncpg://", "postgresql://").replace("postgresql+psycopg://", "postgresql://")
    _pool = AsyncConnectionPool(dsn, min_size=1, max_size=5, kwargs={"autocommit": True})
    await _pool.open()
    _saver = AsyncPostgresSaver(_pool)
    await _saver.setup()  # creates tables on first run (idempotent)
    return _saver
```

"""Central agent registry. Each new agent (02-07, 02-08) registers here."""
from typing import Any, Callable
from langgraph.graph.state import CompiledStateGraph

# graph_name → factory(checkpointer) → CompiledStateGraph
GRAPH_REGISTRY: dict[str, Callable[[Any], CompiledStateGraph]] = {}


def register(name: str):
    """Decorator: @register('my-agent')."""
    def deco(factory):
        GRAPH_REGISTRY[name] = factory
        return factory
    return deco
```

**`app/graphs/echo_with_hitl.py`** : minimal demo agent showing HITL pattern.
```python
"""Demo agent: echo text, but pause for human approval before final output."""
from typing import TypedDict
from langgraph.graph import StateGraph, START, END
from app.graphs.registry import register


class EchoState(TypedDict):
    user_text: str
    draft: str
    approved: bool | None
    final: str


def draft_node(state: EchoState) -> EchoState:
    return {**state, "draft": f"Echo: {state['user_text']}"}


def await_approval_node(state: EchoState) -> EchoState:
    # No-op: this node exists so we can interrupt_before it
    # Resume call with state.approved=true|false will pass through here
    return state


def finalize_node(state: EchoState) -> EchoState:
    if state.get("approved"):
        return {**state, "final": state["draft"]}
    return {**state, "final": "(rejected by human)"}


@register("echo-with-hitl")
def make_graph(checkpointer):
    g = StateGraph(EchoState)
    g.add_node("draft", draft_node)
    g.add_node("await_approval", await_approval_node)
    g.add_node("finalize", finalize_node)
    g.add_edge(START, "draft")
    g.add_edge("draft", "await_approval")
    g.add_edge("await_approval", "finalize")
    g.add_edge("finalize", END)
    return g.compile(checkpointer=checkpointer, interrupt_before=["await_approval"])
```

**`app/routes/agents.py`** :
```python
from typing import Any
from uuid import uuid4
from fastapi import APIRouter, Depends, HTTPException
from pydantic import BaseModel, ConfigDict
from app.checkpointer import get_checkpointer
from app.graphs.registry import GRAPH_REGISTRY
from app.deps import get_current_principal, get_team_scope

router = APIRouter()


class RunBody(BaseModel):
    model_config = ConfigDict(extra="forbid")
    initial_state: dict[str, Any]
    thread_id: str | None = None


@router.post("/agents/{agent_name}/run")
async def run_agent(agent_name: str, body: RunBody,
                    principal: dict = Depends(get_current_principal),
                    team_scope: str = Depends(get_team_scope)):
    if agent_name not in GRAPH_REGISTRY:
        raise HTTPException(404, f"unknown agent {agent_name}")
    cp = await get_checkpointer()
    graph = GRAPH_REGISTRY[agent_name](cp)
    thread_id = body.thread_id or str(uuid4())
    config = {"configurable": {"thread_id": thread_id, "team_scope": team_scope}}
    events = []
    async for event in graph.astream(body.initial_state, config):
        events.append(event)
    state = await graph.aget_state(config)
    return {
        "thread_id": thread_id,
        "agent": agent_name,
        "state": state.values if hasattr(state, "values") else state,
        "interrupted_at": list(state.next) if hasattr(state, "next") and state.next else None,
        "events_count": len(events),
    }


@router.get("/agents/{thread_id}/state")
async def get_state(thread_id: str, agent_name: str | None = None,
                   team_scope: str = Depends(get_team_scope)):
    if not agent_name or agent_name not in GRAPH_REGISTRY:
        raise HTTPException(400, "agent_name required")
    cp = await get_checkpointer()
    graph = GRAPH_REGISTRY[agent_name](cp)
    config = {"configurable": {"thread_id": thread_id}}
    state = await graph.aget_state(config)
    if state is None:
        raise HTTPException(404)
    return {"thread_id": thread_id, "state": state.values if hasattr(state, "values") else state,
            "next": list(state.next) if hasattr(state, "next") else []}


class ResumeBody(BaseModel):
    model_config = ConfigDict(extra="forbid")
    agent_name: str
    state_update: dict[str, Any] = {}


@router.post("/agents/{thread_id}/resume")
async def resume_agent(thread_id: str, body: ResumeBody,
                       team_scope: str = Depends(get_team_scope)):
    if body.agent_name not in GRAPH_REGISTRY:
        raise HTTPException(404)
    cp = await get_checkpointer()
    graph = GRAPH_REGISTRY[body.agent_name](cp)
    config = {"configurable": {"thread_id": thread_id, "team_scope": team_scope}}
    if body.state_update:
        await graph.aupdate_state(config, body.state_update)
    events = []
    async for event in graph.astream(None, config):
        events.append(event)
    state = await graph.aget_state(config)
    return {"thread_id": thread_id, "state": state.values, "events_count": len(events)}
```
