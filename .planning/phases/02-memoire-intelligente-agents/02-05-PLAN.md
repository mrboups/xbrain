**`app/services/rag_enrichment.py`** :
```python
"""RAG enrichment: build a system_addendum markdown from CANONICAL facts."""

from xbrain_memory import MemoryProvider, TruthLevel, SearchHit

DEFAULT_TOP_K = 5
MAX_FACT_CHARS = 280  # Twitter-length cap per fact, keeps prompt small


async def build_system_addendum(
    provider: MemoryProvider,
    *,
    query: str,
    team_scope: str,
    project_scope: str | None = None,
    top_k: int = DEFAULT_TOP_K,
    min_level: TruthLevel = TruthLevel.CANONICAL,
) -> str:
    """Returns markdown addendum (or empty string if no facts found).

    Selected facts must satisfy:
      - team_scope == team_scope (mandatory)
      - project_scope == project_scope (if specified)
      - truth_level >= min_level (default CANONICAL)
    """
    hits: list[SearchHit] = await provider.search(
        query=query,
        team_scope=team_scope,
        project_scope=project_scope,
        truth_level_min=min_level,
        limit=top_k,
    )
    if not hits:
        return ""
    lines = [
        "## Team facts ({lvl}+ truth level)".format(lvl=min_level.value),
        "",
        "Use these as ground truth in your response. Cite the fact ID if you reference one.",
        "",
    ]
    for h in hits:
        content = h.item.content[:MAX_FACT_CHARS]
        if len(h.item.content) > MAX_FACT_CHARS:
            content += "…"
        lines.append(f"- ({h.item.id[:8]}, conf={h.item.confidence:.2f}): {content}")
    return "\n".join(lines)
```

**`app/routes/system_prompt.py`** :
```python
from fastapi import APIRouter, Depends, Query
from pydantic import BaseModel
from xbrain_memory import MemoryProvider, TruthLevel
from app.deps import get_team_scope, get_memory_provider
from app.services.rag_enrichment import build_system_addendum, DEFAULT_TOP_K

router = APIRouter()


class SystemPromptOut(BaseModel):
    system_addendum: str
    fact_count: int


@router.get("/system-prompt", response_model=SystemPromptOut)
async def get_system_prompt(
    query: str = Query(..., min_length=1, max_length=500),
    project_scope: str | None = Query(default=None),
    top_k: int = Query(default=DEFAULT_TOP_K, ge=1, le=20),
    min_level: TruthLevel = Query(default=TruthLevel.CANONICAL),
    team_scope: str = Depends(get_team_scope),
    provider: MemoryProvider = Depends(get_memory_provider),
):
    addendum = await build_system_addendum(
        provider, query=query, team_scope=team_scope,
        project_scope=project_scope, top_k=top_k, min_level=min_level,
    )
    fact_count = addendum.count("- (")
    return SystemPromptOut(system_addendum=addendum, fact_count=fact_count)
```

**`apps/librechat-bridge/app/conv_enricher.py`** :
```python
"""Inject xbrain CANONICAL facts as system message at the start of new LibreChat conversations.

Triggered by a 'conversations' insert in Mongo change stream (parallel to messages watcher).
Idempotent via metadata flag conversations.metadata.xbrain_enriched=true after injection.
"""
import structlog
from app.memory_api_client import MemoryApiClient

log = structlog.get_logger()


async def enrich_new_conversation(conv_doc: dict, db, mem: MemoryApiClient,
                                  *, sub: str, team_scope: str) -> bool:
    """Process a Mongo `conversations` insert event. Returns True if enriched."""
    conv_id = conv_doc.get("conversationId")
    if not conv_id:
        return False
    # Idempotency: skip if already enriched
    md = conv_doc.get("metadata") or {}
    if md.get("xbrain_enriched"):
        return False
    title = conv_doc.get("title") or ""
    if not title:
        # New conversations may not have a title yet; use first message later
        return False

    # Fetch RAG addendum from memory-api
    sys = await mem.get_system_prompt(sub=sub, team_scope=team_scope, query=title)
    addendum = sys.get("system_addendum", "")
    if not addendum:
        log.info("no_canonical_facts_for_team", team=team_scope)
        # Still mark enriched=true to avoid re-checking
        await db["conversations"].update_one(
            {"conversationId": conv_id},
            {"$set": {"metadata.xbrain_enriched": True}}
        )
        return False

    # Inject as system message
    sys_msg = {
        "conversationId": conv_id,
        "messageId": f"xbrain-system-{conv_id}",
        "user": "system",
        "isCreatedByUser": False,
        "text": addendum,
        "model": conv_doc.get("model"),
        "metadata": {"xbrain_injected": True, "source": "memory-api:rag-canonical"},
        "createdAt": conv_doc.get("createdAt"),
    }
    await db["messages"].insert_one(sys_msg)
    await db["conversations"].update_one(
        {"conversationId": conv_id},
        {"$set": {"metadata.xbrain_enriched": True, "metadata.xbrain_facts_injected":
                  sys.get("fact_count", 0)}}
    )
    log.info("conv_enriched", conv=conv_id, facts=sys.get("fact_count", 0))
    return True
```
