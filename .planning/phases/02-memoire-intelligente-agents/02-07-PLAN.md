```python
"""Document loaders: URL (HTTP fetch + readability) and PDF (pypdf parse)."""
import httpx
from io import BytesIO

async def load_url(url: str) -> str:
    async with httpx.AsyncClient(timeout=30, follow_redirects=True) as c:
        r = await c.get(url)
        r.raise_for_status()
        # Naive: return raw text. Phase 3 = readability/trafilatura.
        return r.text[:50000]  # cap 50KB to avoid huge contexts


async def load_pdf_bytes(content: bytes) -> str:
    from pypdf import PdfReader
    reader = PdfReader(BytesIO(content))
    pages = [p.extract_text() or "" for p in reader.pages]
    return "\n\n".join(pages)[:50000]
```

```python
"""LLM-based fact extraction via Claude."""
import json
from typing import Any
from anthropic import AsyncAnthropic
from app.config import settings

_client: AsyncAnthropic | None = None
def _get_client() -> AsyncAnthropic:
    global _client
    if _client is None:
        if not settings.ANTHROPIC_API_KEY:
            raise RuntimeError("ANTHROPIC_API_KEY required for ingestion-agent")
        _client = AsyncAnthropic(api_key=settings.ANTHROPIC_API_KEY)
    return _client


SYSTEM_PROMPT = """You are a fact extractor for the xbrain memory system.

Extract atomic, verifiable facts from the provided document. Each fact must:
1. Be self-contained (understandable without the source doc)
2. Be verifiable (states something testable)
3. Avoid opinions/speculation
4. Have suggested_truth_level: WORKING (default), VALIDATED (if doc is canonical org doc), or EPHEMERAL (if uncertain)
5. Have confidence: 0.0-1.0 (how certain is THIS extraction, not the fact's truthiness)

Return strict JSON: {"facts": [{"content": "...", "suggested_truth_level": "WORKING|VALIDATED|EPHEMERAL", "confidence": 0.85}]}

Max 20 facts. Quality over quantity."""


async def extract_facts(document_text: str, source_url: str) -> list[dict[str, Any]]:
    client = _get_client()
    msg = await client.messages.create(
        model="claude-3-5-sonnet-latest",
        max_tokens=4096,
        system=SYSTEM_PROMPT,
        messages=[{"role": "user", "content": f"Source: {source_url}\n\n---\n\n{document_text}"}],
    )
    text = "".join(b.text for b in msg.content if b.type == "text")
    try:
        data = json.loads(text)
        facts = data.get("facts", [])
    except json.JSONDecodeError:
        # Try extracting JSON from markdown code block
        import re
        m = re.search(r"```(?:json)?\s*(\{.*?\})\s*```", text, re.DOTALL)
        if m:
            data = json.loads(m.group(1))
            facts = data.get("facts", [])
        else:
            facts = []
    # Validate + add source url
    return [
        {
            "content": f["content"][:500],
            "suggested_truth_level": f.get("suggested_truth_level", "WORKING"),
            "confidence": float(f.get("confidence", 0.7)),
            "source": f"agent:ingestion-v1",
            "metadata": {"source_url": source_url},
        }
        for f in facts
        if isinstance(f, dict) and f.get("content")
    ][:20]
```

```python
"""Ingestion agent: URL/PDF → extract facts → HITL approve → write to memory."""
from typing import Any, TypedDict
from langgraph.graph import StateGraph, START, END
from app.graphs.registry import register
from app.tools.document_loader import load_url, load_pdf_bytes
from app.tools.extract_facts import extract_facts
from app.memory_client import MemoryApiClient

_mem = MemoryApiClient()


class IngestionState(TypedDict):
    source_type: str          # "url" | "pdf"
    source_url: str | None
    pdf_b64: str | None
    document_text: str
    extracted_facts: list[dict[str, Any]]
    approved_indexes: list[int] | None  # filled by HITL — which facts to keep
    written_ids: list[str]
    team_scope: str
    sub: str  # OIDC sub of the proposing user


async def load_node(state: IngestionState) -> IngestionState:
    if state["source_type"] == "url":
        text = await load_url(state["source_url"])
    elif state["source_type"] == "pdf":
        import base64
        text = await load_pdf_bytes(base64.b64decode(state["pdf_b64"]))
    else:
        raise ValueError("source_type must be url|pdf")
    return {**state, "document_text": text}


async def extract_node(state: IngestionState) -> IngestionState:
    facts = await extract_facts(state["document_text"], state.get("source_url") or "pdf-upload")
    return {**state, "extracted_facts": facts}


async def await_review_node(state: IngestionState) -> IngestionState:
    # No-op: interrupt_before this node. Resume call passes approved_indexes.
    return state


async def write_node(state: IngestionState) -> IngestionState:
    approved = state.get("approved_indexes") or list(range(len(state["extracted_facts"])))
    written = []
    for idx in approved:
        if idx < 0 or idx >= len(state["extracted_facts"]):
            continue
        f = state["extracted_facts"][idx]
        item_id = await _mem.upsert_memory(
            sub=state["sub"], team_scope=state["team_scope"],
            content=f["content"],
            truth_level=f.get("suggested_truth_level", "WORKING"),
            confidence=f.get("confidence", 0.7),
            source=f.get("source", "agent:ingestion-v1"),
            metadata={**f.get("metadata", {}), "ingestion_thread": state.get("_thread_id", "?")},
        )
        written.append(item_id)
    return {**state, "written_ids": written}


@register("ingestion")
def make_graph(checkpointer):
    g = StateGraph(IngestionState)
    g.add_node("load", load_node)
    g.add_node("extract", extract_node)
    g.add_node("await_review", await_review_node)
    g.add_node("write", write_node)
    g.add_edge(START, "load")
    g.add_edge("load", "extract")
    g.add_edge("extract", "await_review")
    g.add_edge("await_review", "write")
    g.add_edge("write", END)
    return g.compile(checkpointer=checkpointer, interrupt_before=["await_review"])
```
