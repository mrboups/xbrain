"""Second opinion agent: ask Claude AND Grok in parallel, return diff."""
import asyncio
from typing import TypedDict
from anthropic import AsyncAnthropic
from openai import AsyncOpenAI  # used for xAI Grok via OpenAI-compat API
from langgraph.graph import StateGraph, START, END
from app.graphs.registry import register
from app.config import settings


class SecondOpinionState(TypedDict):
    prompt: str
    claude_response: str
    grok_response: str
    claude_error: str | None
    grok_error: str | None
    diff_highlights: str
    final_markdown: str


async def call_claude(prompt: str) -> tuple[str, str | None]:
    if not settings.ANTHROPIC_API_KEY:
        return "", "ANTHROPIC_API_KEY not configured"
    try:
        c = AsyncAnthropic(api_key=settings.ANTHROPIC_API_KEY)
        r = await c.messages.create(
            model="claude-3-5-sonnet-latest",
            max_tokens=2048,
            messages=[{"role": "user", "content": prompt}],
        )
        return "".join(b.text for b in r.content if b.type == "text"), None
    except Exception as e:
        return "", f"{type(e).__name__}: {e}"


async def call_grok(prompt: str) -> tuple[str, str | None]:
    if not getattr(settings, "XAI_API_KEY", None):
        return "", "XAI_API_KEY not configured"
    try:
        c = AsyncOpenAI(api_key=settings.XAI_API_KEY, base_url="https://api.x.ai/v1")
        r = await c.chat.completions.create(
            model="grok-2-latest",
            messages=[{"role": "user", "content": prompt}],
            max_tokens=2048,
        )
        return r.choices[0].message.content or "", None
    except Exception as e:
        return "", f"{type(e).__name__}: {e}"


async def parallel_call_node(state: SecondOpinionState) -> SecondOpinionState:
    (claude_text, claude_err), (grok_text, grok_err) = await asyncio.gather(
        call_claude(state["prompt"]), call_grok(state["prompt"]),
    )
    return {
        **state,
        "claude_response": claude_text,
        "grok_response": grok_text,
        "claude_error": claude_err,
        "grok_error": grok_err,
    }


async def diff_node(state: SecondOpinionState) -> SecondOpinionState:
    """Compute simple diff highlights — Phase 2 = naive (length, sentiment-ish keywords).
    Phase 3 = use Claude as a judge to point divergences."""
    c, g = state["claude_response"], state["grok_response"]
    if not c or not g:
        return {**state, "diff_highlights": "(can't diff — one or both responses missing)"}
    bullets = []
    if abs(len(c) - len(g)) > 200:
        bullets.append(f"- Length differs significantly: Claude {len(c)} chars, Grok {len(g)} chars")
    # Naive disagreement keywords
    disagree_words = ["however", "actually", "wrong", "incorrect", "but"]
    c_disagree = sum(1 for w in disagree_words if w in c.lower())
    g_disagree = sum(1 for w in disagree_words if w in g.lower())
    if abs(c_disagree - g_disagree) >= 2:
        bullets.append(f"- Hedging/contrast words differ: Claude={c_disagree}, Grok={g_disagree}")
    if not bullets:
        bullets.append("- No obvious surface-level divergences (read both for nuance)")
    return {**state, "diff_highlights": "\n".join(bullets)}


async def format_node(state: SecondOpinionState) -> SecondOpinionState:
    parts = ["## Claude (claude-3-5-sonnet-latest)\n"]
    if state["claude_error"]:
        parts.append(f"_unavailable: {state['claude_error']}_\n")
    else:
        parts.append(state["claude_response"])
    parts += ["\n\n## Grok (grok-2-latest)\n"]
    if state["grok_error"]:
        parts.append(f"_unavailable: {state['grok_error']}_\n")
    else:
        parts.append(state["grok_response"])
    parts += ["\n\n## Diff highlights\n", state["diff_highlights"]]
    return {**state, "final_markdown": "".join(parts)}


@register("second-opinion")
def make_graph(checkpointer):
    g = StateGraph(SecondOpinionState)
    g.add_node("parallel", parallel_call_node)
    g.add_node("diff", diff_node)
    g.add_node("format", format_node)
    g.add_edge(START, "parallel")
    g.add_edge("parallel", "diff")
    g.add_edge("diff", "format")
    g.add_edge("format", END)
    return g.compile(checkpointer=checkpointer)
