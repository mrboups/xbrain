class OrgBlockBody(BaseModel):

@router.post("/teams/{team_id}/org-blocks", response_model=OrgBlockOut, status_code=201)
async def create_org_block(
    team_id: UUID,
    body: OrgBlockBody,
    principal: dict[str, Any] = Depends(get_current_principal),
    session: AsyncSession = Depends(get_session),

    """Pre-block a GitHub login from joining this team via auto-grant.
    Takes effect even before the user signs into xbrain."""

    if team is None:
        raise HTTPException(404, "team not found")
    actor = await _require_team_admin(principal, team, session)

        session, team_id=team_id, github_login=body.github_login, blocked_by=actor.id

    await write_audit(
        session,
        actor_user_id=actor.id,
        team_scope=team.slug,
        action="org_block.create",
        target_id=body.github_login,
        payload={},

        raise HTTPException(500, "block disappeared after insert")
    return OrgBlockOut(
        github_login=block.github_login,
        blocked_at=block.blocked_at.isoformat(),
        blocked_by_email=actor.email,

### Task 4 — Wire `get_team_scope` to enforce blocked_at in BOTH principal branches (commit #4)

**REVISION 1 — B-3 fix.** The original plan added the `blocked_at` 403 check
ONLY in the `kind == "user"` branch of `get_team_scope`. But `get_team_scope`
also has a `kind == "user_api_token"` branch (the `xbt_` path) that runs
BEFORE the standard membership lookup — and it returns the `x_team_scope`
header value without consulting `team_members.blocked_at` at all.

That means: a user who minted an `xbt_` token (multi-team, `team_scope=NULL`),
then got blocked by an admin, can keep using that pre-minted token to call any
team-scoped endpoint with `X-Team-Scope: <team>` and the block silently does
nothing. The bypass is total (B-3 BLOCKER).

The fix below adds the `blocked_at` check to BOTH branches.

In `apps/memory-api/app/deps.py`, in `get_team_scope` (currently lines 216-242),
replace the function body to look like this:

```python
async def get_team_scope(
    request: Request,
    x_team_scope: str | None = Header(default=None, alias="X-Team-Scope"),
    principal: dict[str, Any] = Depends(get_current_principal),
    session: AsyncSession = Depends(get_session),
) -> str:
    """Resolve the effective team scope for this request and enforce membership.

    Phase 10 (REVISION 1 B-3): the `user_api_token` branch — which previously
    returned early after verifying token_scope vs header match — now ALSO
    consults team_members.blocked_at for the principal's user_id. Without
    this check, a user blocked AFTER minting an xbt_ token could continue
    using that token to bypass the block on every team-scoped endpoint.
    """
    if x_team_scope is None:
        raise HTTPException(400, "X-Team-Scope header required")

    # === user_api_token (xbt_) branch — REVISION 1 B-3 enforcement ===
    if principal["kind"] == "user_api_token":
        # Token-scope match (existing logic, kept verbatim).
        token_scope = principal.get("api_token_team_scope")
        if token_scope is not None and token_scope != x_team_scope:
            raise HTTPException(
                403,
                f"API token team_scope mismatch with X-Team-Scope header",
            )
        # Phase 10 B-3 — enforce block at the scoped-token boundary.
        # The principal["user"] row is the survivor (follow_merge_pointer
        # already ran in get_current_principal per 10-02 Task 5).
        member = await get_membership(
            session, user_id=principal["user"].id, team_slug=x_team_scope
        )
        if member is not None and member.blocked_at is not None:
            raise HTTPException(403, f"Member blocked from team {x_team_scope}")
        # We do NOT require membership existence in the xbt_ branch — the
        # multi-team token may scope to teams the user hasn't joined, in
        # which case downstream handlers will fail for their own reasons.
        # The block check is the only added enforcement here.
        return x_team_scope

    # === user (gho_ / Google OIDC) branch — Phase 10 GHA-03 enforcement ===
    user = principal["user"]
    membership = await get_membership(
        session, user_id=user.id, team_slug=x_team_scope
    )
    if membership is None:
        raise HTTPException(403, f"Not a member of team {x_team_scope}")
    if membership.blocked_at is not None:
        raise HTTPException(403, f"Member blocked from team {x_team_scope}")
    return x_team_scope
```

**Executor note:** depending on the current shape of `get_team_scope` in deps.py
(which 10-02 may have already modified), preserve any pre-existing logic in
the `user` branch (e.g. github_is_org_member checks). The two essentials are:
1. The `user_api_token` branch MUST query `get_membership` and reject on
   `blocked_at is not None`.
2. The `user` branch keeps its existing membership-required behaviour plus
   the same `blocked_at` 403.
