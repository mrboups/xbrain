```html
    const API_BASE = window.location.origin.includes('localhost')
      ? 'http://localhost:8000'
      : 'https://api.dejavu.cat';
    const POLL_INTERVAL_MS = 30000;
    let pollTimer = null;

    function $(id) { return document.getElementById(id); }

    function getAuthHeaders() {
      const token = $('auth-token').value.trim();
      const team = $('filter-team').value.trim();
      if (!token || !team) return null;
      // Persist to localStorage for convenience
      try { localStorage.setItem('xbrain_token', token); localStorage.setItem('xbrain_team', team); } catch(_){}

        'Authorization': 'Bearer ' + token,
        'X-Team-Scope': team,
      };

    function badgeFor(status) {
      const cls = {
        'todo': 'bg-gray-200 text-gray-800',
        'in_progress': 'bg-blue-200 text-blue-800',
        'done': 'bg-green-200 text-green-800',
        'cancelled': 'bg-red-200 text-red-800',
      }[status] || 'bg-gray-200 text-gray-800';
      return `<span class="text-xs px-2 py-0.5 rounded ${cls}">${status}</span>`;

      return 'priority-' + (p || 'normal');

    function renderTasks(tasks) {
      const root = $('task-list');
      if (tasks.length === 0) {
        root.innerHTML = '<div class="px-4 py-6 text-gray-400 text-sm text-center">No tasks match.</div>';
        return;

            <div class="font-medium">${escapeHtml(t.title)}</div>
            ${t.description ? `<div class="text-sm text-gray-600 mt-1">${escapeHtml(t.description.slice(0, 200))}</div>` : ''}
            <div class="text-xs text-gray-400 mt-1">
              ${t.project_scope ? 'project: ' + escapeHtml(t.project_scope) : ''}

              • created ${new Date(t.created_at).toLocaleString()}
            </div>
          </div>
          <div class="flex flex-col gap-1">
            ${t.status !== 'done' ? `<button data-id="${t.id}" data-action="done" class="task-action text-xs bg-green-600 text-white px-2 py-1 rounded">Mark done</button>` : ''}
            ${t.status === 'todo' ? `<button data-id="${t.id}" data-action="in_progress" class="task-action text-xs bg-blue-600 text-white px-2 py-1 rounded">Start</button>` : ''}
          </div>
        </div>
      `).join('');

      root.querySelectorAll('.task-action').forEach(btn => {
        btn.addEventListener('click', async () => {
          const id = btn.dataset.id;
          const newStatus = btn.dataset.action;
          await patchStatus(id, newStatus);
          await loadTasks();
        });
      });
    }

    function escapeHtml(s) {
      return String(s || '').replace(/[&<>"']/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'})[c]);
    }

    async function loadTasks() {
      const headers = getAuthHeaders();
      if (!headers) {
        $('status-line').textContent = 'Enter team + token then click Refresh.';
        return;
      }
      const status = $('filter-status').value;
      const project = $('filter-project').value;
      const params = new URLSearchParams();
      if (status) params.set('status', status);
      if (project) params.set('project_scope', project);
      params.set('limit', '100');
      try {
        $('status-line').textContent = 'Loading…';
        const r = await fetch(API_BASE + '/v1/tasks?' + params.toString(), { headers });
        if (!r.ok) {
          $('status-line').textContent = `Error ${r.status}: ${await r.text().then(t => t.slice(0, 200))}`;
          return;
        }
        const tasks = await r.json();
        renderTasks(tasks);
        $('status-line').textContent = `${tasks.length} task(s) — last updated ${new Date().toLocaleTimeString()}`;
        // Populate project filter on first load
        const ps = $('filter-project');
        const seen = new Set(Array.from(ps.options).map(o => o.value));
        tasks.forEach(t => {
          if (t.project_scope && !seen.has(t.project_scope)) {
            const opt = document.createElement('option');
            opt.value = t.project_scope; opt.textContent = t.project_scope;
            ps.appendChild(opt); seen.add(t.project_scope);
          }
        });
      } catch (e) {
        $('status-line').textContent = 'Network error: ' + e.message;
      }
    }

    async function patchStatus(taskId, status) {
      const headers = getAuthHeaders();
      if (!headers) return;
      headers['Content-Type'] = 'application/json';
      await fetch(API_BASE + '/v1/tasks/' + taskId, {
        method: 'PATCH',
        headers,
        body: JSON.stringify({ status }),
      });
    }

    function startPolling() {
      if (pollTimer) clearInterval(pollTimer);
      pollTimer = setInterval(loadTasks, POLL_INTERVAL_MS);
    }

    // Initial wiring
    document.addEventListener('DOMContentLoaded', () => {
      try {
        $('auth-token').value = localStorage.getItem('xbrain_token') || '';
        $('filter-team').value = localStorage.getItem('xbrain_team') || '';
      } catch(_) {}
      $('refresh-btn').addEventListener('click', loadTasks);
      $('filter-status').addEventListener('change', loadTasks);
      $('filter-project').addEventListener('change', loadTasks);
      if ($('auth-token').value && $('filter-team').value) {
        loadTasks();
        startPolling();
      }
    });
  </script>
</body>
</html>
```

```bash
#!/usr/bin/env bash
#
# verify-phase7.sh — Verify Phase 7 (CRM + Granola + Task Intelligence) deliverables.
#
# Tests:
#   1. Migration 0010 applied (alembic_version = 0010)
#   2. teams.plan column exists with check constraint
#   3. contacts table exists with type CHECK
#   4. tasks table exists with FK to contacts
#   5. granola_integrations table exists
#   6. Container xbrain-granola-sync is running and healthy
#   7. Endpoint /v1/tasks responds (any auth response, validates routing)
#   8. Endpoint /v1/crm/contacts responds (any auth response)
#
# Usage: bash infrastructure/scripts/verify-phase7.sh

set -uo pipefail   # NOT -e — we want all tests to run even if some fail

TEST_TOTAL=8
TEST_PASSED=0
FAILURES=()

DB_CONTAINER="${DB_CONTAINER:-xbrain-postgres}"
GRANOLA_CONTAINER="${GRANOLA_CONTAINER:-xbrain-granola-sync}"
MEMAPI_HOST="${MEMAPI_HOST:-https://api.dejavu.cat}"

pass() { echo "  PASS: $1"; TEST_PASSED=$((TEST_PASSED + 1)); }
fail() { echo "  FAIL: $1 — $2"; FAILURES+=("$1: $2"); }

run_psql() {
  docker exec -i "$DB_CONTAINER" psql -U xbrain -d xbrain -tAc "$1" 2>/dev/null
}

echo "=== Phase 7 Verification ==="

# Test 1: alembic_version = 0010
echo
echo "[1/8] Migration 0010 applied"
ver=$(run_psql "SELECT version_num FROM alembic_version" || true)
if [ "$ver" = "0010" ]; then
  pass "alembic_version = 0010"
else
  fail "Migration not at 0010" "got: '$ver'"
fi

# Test 2: teams.plan column with CHECK constraint
echo
echo "[2/8] teams.plan column + CHECK"
plan_col=$(run_psql "SELECT column_name FROM information_schema.columns WHERE table_name='teams' AND column_name='plan'" || true)
plan_chk=$(run_psql "SELECT conname FROM pg_constraint WHERE conname='teams_plan_check'" || true)
if [ "$plan_col" = "plan" ] && [ "$plan_chk" = "teams_plan_check" ]; then
  pass "teams.plan column + CHECK present"
else
  fail "teams.plan missing" "col='$plan_col' chk='$plan_chk'"
fi

# Test 3: contacts table
echo
echo "[3/8] contacts table + type CHECK"
contacts_tbl=$(run_psql "SELECT to_regclass('contacts')::text" || true)
contacts_chk=$(run_psql "SELECT conname FROM pg_constraint WHERE conname='contacts_type_check'" || true)
if [ "$contacts_tbl" = "contacts" ] && [ "$contacts_chk" = "contacts_type_check" ]; then
  pass "contacts table + CHECK present"
else
  fail "contacts table missing" "tbl='$contacts_tbl' chk='$contacts_chk'"
fi

# Test 4: tasks table with FK to contacts
echo
echo "[4/8] tasks table + FK to contacts"
tasks_tbl=$(run_psql "SELECT to_regclass('tasks')::text" || true)
tasks_fk=$(run_psql "SELECT conname FROM pg_constraint WHERE conrelid='tasks'::regclass AND confrelid='contacts'::regclass LIMIT 1" || true)
if [ "$tasks_tbl" = "tasks" ] && [ -n "$tasks_fk" ]; then
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Browser → tasks.html (Firebase Hosting) | Static page, no server logic. XSS protection via escapeHtml on all dynamic content. |
| tasks.html ↔ memory-api | CORS request from Firebase domain to api.dejavu.cat. Auth via Bearer token + X-Team-Scope. |
| Token in localStorage | Persistence convenience only. Compromised browser = compromised token. v1 acceptable; deferred to user education. |
| Nginx public ↔ memory-api internal | Nginx is the trust boundary. Authorization header proxied through unchanged. |
| verify-phase7.sh ↔ docker socket | Script runs locally on VM with docker access. Operator-only path. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-07-07-01 | XSS | tasks.html rendering task content | mitigate | All user-controlled fields (title, description, project_scope, assigned_to UUID-prefix) wrapped in `escapeHtml()` before HTML insertion. innerHTML of escaped content is safe. |
| T-07-07-02 | Token theft | Bearer token in localStorage | accept | Standard SPA pattern. Memory-only storage (sessionStorage) prefered but localStorage is OK for this internal tool. v2: migrate to httpOnly cookie. |
| T-07-07-03 | CSRF | tasks.html POSTing to api.dejavu.cat | mitigate | API requires `Authorization: Bearer` (custom header) — browser does NOT auto-include from cross-origin form POSTs. CSRF impractical without token leak. |
| T-07-07-04 | Information Disclosure | Nginx logs include Authorization header | mitigate | Default nginx access logs do NOT include request headers. Audit: confirm `log_format` in conf does not include `$http_authorization`. |
| T-07-07-05 | Spoofing | tasks.html accepting arbitrary X-Team-Scope value | accept | Server-side validation (memory-api `get_team_scope`) confirms membership before honoring. Client-side header value is just a hint. |
| T-07-07-06 | Denial of Service | Polling 30s × N users → memory-api load | accept | Polling is per-user, per-tab. memory-api can handle 100s of req/s. SSE migration deferred (research recommendation). |
