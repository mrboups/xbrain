**memory_api_client.py** — bridge JWT signer + upsert (matches granola-sync pattern):
```python
import time
from typing import Any
import httpx
import structlog
from authlib.jose import jwt as jose_jwt
from app.config import settings

log = structlog.get_logger(__name__)


def _make_bridge_jwt(acting_user_sub: str) -> str:
    payload = {
        "sub": "session-bridge",
        "iss": "session-bridge",
        "scope": "bridge",
        "acting_user_sub": acting_user_sub,
        "iat": int(time.time()),
        "exp": int(time.time()) + 3600,
    }
    return jose_jwt.encode(
        {"alg": settings.JWT_ALGORITHM},
        payload,
        settings.BRIDGE_SHARED_SECRET,
    ).decode()


async def upsert_external_session(
    user_sub: str,
    provider: str,
    extension_id: str | None,
    metadata: dict[str, Any] | None,
) -> bool:
    """POST to memory-api /v1/me/external-sessions. Returns True on 2xx, False otherwise.
    Fire-and-forget: never raises (caller wraps in asyncio.create_task)."""
    if not settings.BRIDGE_SHARED_SECRET:
        log.warning("register.upsert.skipped", reason="no_bridge_secret")
        return False
    token = _make_bridge_jwt(user_sub)
    body = {"provider": provider, "extension_id": extension_id, "metadata": metadata or {}}
    try:
        async with httpx.AsyncClient(timeout=5.0) as client:
            r = await client.post(
                f"{settings.MEMORY_API_URL}/v1/me/external-sessions",
                json=body,
                headers={"Authorization": f"Bearer {token}", "Content-Type": "application/json"},
            )
        if 200 <= r.status_code < 300:
            log.info("register.upsert.ok", sub=user_sub, provider=provider, status=r.status_code)
            return True
        log.warning("register.upsert.failed", sub=user_sub, provider=provider, status=r.status_code, body=r.text[:200])
        return False
    except Exception as e:
        log.warning("register.upsert.error", sub=user_sub, provider=provider, err=str(e))
        return False
```

**pool.py** — class with asyncio.Lock guarding state (per 09-RESEARCH "Don't Hand-Roll" row about FastAPI WS registry):
```python
class Pool:
    def __init__(self):
        self._lock = asyncio.Lock()
        self._sockets: dict[str, WebSocket] = {}
        self._queues: dict[str, dict[str, asyncio.Queue]] = {}

    async def register(self, sub, ws) -> WebSocket | None:  # returns previous WS if any (caller closes it)
    async def unregister(self, sub, ws) -> None:  # only unregisters if current entry == ws
    async def push_request(self, sub, request_id, openai_body) -> asyncio.Queue:  # raises KeyError if no socket; creates per-request queue and ws.send_json
    async def deliver_chunk(self, sub, request_id, msg) -> None:  # puts msg on queue if exists; drops silently otherwise
    async def drain(self, sub, reason: str) -> None:  # pushes {"type":"error","detail":reason} to all queues for sub, then drops the sub's queue dict
    def active_count(self) -> int  # for /metrics

pool = Pool()  # module singleton
```
