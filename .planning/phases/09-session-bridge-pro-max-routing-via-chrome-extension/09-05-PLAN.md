```javascript
// === Phase 9: Sessions section ===
const MEMORY_API_BASE = "https://api.grooveos.app";

async function renderSessions() {
  await renderWsStatus();
  await renderClaudeSessionInfo();
}

async function renderWsStatus() {
  const dot = document.getElementById("ws-dot");
  if (!dot) return;
  try {
    const resp = await chrome.runtime.sendMessage({ kind: "ws_status_query" });
    // readyState: 0=CONNECTING 1=OPEN 2=CLOSING 3=CLOSED -1=no ws
    dot.classList.remove("active", "idle");
    if (resp && resp.readyState === 1) {
      dot.classList.add("active");
      dot.title = "WebSocket connected to bridge";
    } else {
      dot.classList.add("idle");
      dot.title = `WebSocket state: ${resp ? resp.readyState : "unknown"}`;
    }
  } catch (e) {
    dot.classList.add("idle");
    dot.title = `WS query failed: ${e.message}`;
  }
}

async function renderClaudeSessionInfo() {
  const email = document.getElementById("claude-email");
  const last = document.getElementById("claude-last-seen");
  if (!email || !last) return;
  const { xbt_token } = await chrome.storage.session.get(["xbt_token"]);
  if (!xbt_token) {
    email.textContent = "non connecté";
    last.textContent = "";
    return;
  }
  try {
    const r = await fetch(`${MEMORY_API_BASE}/v1/me/external-sessions`, {
      headers: { Authorization: `Bearer ${xbt_token}` },
    });
    if (!r.ok) {
      email.textContent = `erreur ${r.status}`;
      return;
    }
    const sessions = await r.json();
    const claude = sessions.find((s) => s.provider === "claude");
    if (!claude) {
      email.textContent = "non détecté";
      last.textContent = "Login to claude.ai in this browser";
      return;
    }
    email.textContent = (claude.metadata && claude.metadata.email_logged) || "—";
    last.textContent = `Last seen: ${formatRelative(claude.last_seen_at)}`;
  } catch (e) {
    email.textContent = `erreur réseau: ${e.message}`;
  }
}

function formatRelative(isoStamp) {
  if (!isoStamp) return "—";
  const then = new Date(isoStamp);
  const diff = (Date.now() - then.getTime()) / 1000;
  if (diff < 60) return `${Math.floor(diff)}s ago`;
  if (diff < 3600) return `${Math.floor(diff / 60)}m ago`;
  if (diff < 86400) return `${Math.floor(diff / 3600)}h ago`;
  return then.toLocaleDateString();
}

async function disconnectClaude() {
  const { xbt_token } = await chrome.storage.session.get(["xbt_token"]);
  if (!xbt_token) return;
  const r = await fetch(`${MEMORY_API_BASE}/v1/me/external-sessions/claude`, {
    method: "DELETE",
    headers: { Authorization: `Bearer ${xbt_token}` },
  });
  if (r.ok || r.status === 404) {
    await renderSessions();
  } else {
    alert(`Disconnect failed: ${r.status}`);
  }
}

document.addEventListener("DOMContentLoaded", () => {
  const btnRefresh = document.getElementById("btn-refresh-claude");
  const btnDisc = document.getElementById("btn-disconnect-claude");
  if (btnRefresh) btnRefresh.addEventListener("click", renderSessions);
  if (btnDisc) btnDisc.addEventListener("click", () => {
    if (confirm("Disconnect Claude session? You'll need to log into claude.ai again.")) {
      disconnectClaude();
    }
  });
  renderSessions();
});
```

- Use `chrome.runtime.sendMessage`, NOT `window.postMessage` (T-09-03-02 mitigation).
